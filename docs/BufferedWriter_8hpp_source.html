125 void write(const Type_* input, std::size_t number) {
126 const auto buffer_size = my_buffer.size();
127
128 // Seeing if we fill up the rest of our buffer or not.
129 const std::size_t remaining = buffer_size - my_current;
130 if (number < remaining) {
131 std::copy_n(input, number, my_buffer.data() + my_current);
132 my_current += number;
133 return;
134 }
135
136 std::copy_n(input, remaining, my_buffer.data() + my_current);
137 number -= remaining;
138 input += remaining;
139
140 if (number == 0) {
141 flush_async();
142 safe_update_position(my_overall, buffer_size, my_overall_err);
143 my_current = 0;
144 return;
145 }
146
147 flush_sync(my_buffer.data(), buffer_size);
148 safe_update_position(my_overall, buffer_size, my_overall_err);
149
150 // Directly filling the output array, bypassing our buffer.
151 while (number >= buffer_size) {
152 flush_sync(input, buffer_size);
153 safe_update_position(my_overall, buffer_size, my_overall_err);
154 input += buffer_size;
155 number -= buffer_size;
156 }
157
158 // Filling the cache and copying the remainder into the output array.
159 std::copy_n(input, number, my_buffer.data());
160 my_current = number;
161 }

322 while (1) {

324 my_cv.wait(lck, [&]() { return my_ready_input; });
325 my_ready_input = false;

347 void flush_async() {
348 auto& buffer = this->get_buffer();
349 const auto num = buffer.size();
350
351 if (my_worker_active) {
352 // If the worker is active, it's probably already gotten started so we just wait for it to finish.
353 // Then we swap the results with the main buffer and submit a new job to the worker.
354 std::unique_lock lck(my_mut);
355 my_cv.wait(lck, [&]() { return my_ready_output; });
356 my_ready_output = false;
357
358 if (my_thread_err) {
359 std::rethrow_exception(my_thread_err);
360 }
361
362 buffer.swap(my_buffer_worker);
363 my_to_write = num;
364
365 my_ready_input = true;
366 lck.unlock();
367 my_cv.notify_one();

369 } else {

371 std::unique_lock lck(my_mut);
372 buffer.swap(my_buffer_worker);

382 void flush_sync(const Type_* ptr, std::size_t num) {
383 if (my_worker_active) {
384 // If the worker is active, we wait for it to finish and then write from the supplied pointer.
385 // We do not submit a new job, based on the loop in the BufferedWriter::write() overload:
386 //
387 // - We'll probably want to call this flush() overload again.
388 // On subsequent calls, we'll want to directly write from the supplied pointer to the Writer to cut out a copy.
389 // - If we didn't call this overload again, we'd probably call the other refill() overload immediately,
390 // So if we sent a new job to the worker, we'd end up immediately blocking on it to submit the next write.
391 // So we might as well skip that communication overhead and write directly on the main thread.
392 std::unique_lock lck(my_mut);
393 my_cv.wait(lck, [&]() { return my_ready_output; });
394 my_ready_output = false;
395 if (my_thread_err) {
396 std::rethrow_exception(my_thread_err);
397 }
398
399 my_writer->write(reinterpret_cast<const unsigned char*>(ptr), num);
400 my_worker_active = false;
401
402 } else {
403 // If the worker isn't active, we can directly write from the supplied pointer, avoiding some communication overhead.
404 my_writer->write(reinterpret_cast<const unsigned char*>(ptr), num);
405 }
